let globalOptions = null; // Modifica la funzione caricaListaUtenti esistente per salvare anche le opzioni globali async function caricaListaUtenti() { const selectUser = document.getElementById('loginUser'); try { const response = await fetch(SCRIPT_URL); const data = await response.json(); // Salva le opzioni (cinema, responsabilità, motivazioni) if (data.options) { globalOptions = data.options; popolaCampiForm(); } selectUser.innerHTML = ''; if (data.users && data.users.length > 0) { data.users.forEach(user => { const opt = document.createElement('option'); opt.value = user; opt.textContent = user; selectUser.appendChild(opt); }); } } catch (err) { console.error("Errore nel caricamento:", err); } } function popolaCampiForm() { if (!globalOptions) return; // Popola Cinema const selCinema = document.getElementById('selCinema'); selCinema.innerHTML = ''; globalOptions.cinemas.forEach(c => { const opt = document.createElement('option'); opt.value = c.name; opt.textContent = c.name; selCinema.appendChild(opt); }); // Popola Responsabilità const selResp = document.getElementById('selResponsibility'); selResp.innerHTML = ''; globalOptions.responsibilities.forEach(r => { const opt = document.createElement('option'); opt.value = r; opt.textContent = r; selResp.appendChild(opt); }); // Popola Motivazioni const selReason = document.getElementById('selReason'); selReason.innerHTML = ''; globalOptions.reasons.forEach(reason => { const opt = document.createElement('option'); opt.value = reason; opt.textContent = reason; selReason.appendChild(opt); }); } function aggiornaDatiCinema() { const nomeCinema = document.getElementById('selCinema').value; if (!globalOptions || !nomeCinema) { document.getElementById('txtScreens').value = ''; document.getElementById('txtCapacity').value = ''; document.getElementById('txtArea').value = ''; return; } const cinemaObj = globalOptions.cinemas.find(c => c.name === nomeCinema); if (cinemaObj) { document.getElementById('txtScreens').value = cinemaObj.screens; document.getElementById('txtCapacity').value = cinemaObj.capacity; document.getElementById('txtArea').value = cinemaObj.area; } } async function salvaReport() { const msgEl = document.getElementById('saveMessage'); const btn = document.getElementById('btnSave'); const reportData = { cinema: document.getElementById('selCinema').value, nScreens: document.getElementById('txtScreens').value, cinemaCapacity: document.getElementById('txtCapacity').value, area: document.getElementById('txtArea').value, responsibility: document.getElementById('selResponsibility').value, reason: document.getElementById('selReason').value, referenceNumber: document.getElementById('txtReference').value, comments: document.getElementById('txtComments').value, compiler: currentUser ? currentUser.username : 'Admin' }; msgEl.style.color = '#0d6efd'; msgEl.textContent = 'Salvataggio in corso...'; btn.disabled = true; try { const response = await fetch(SCRIPT_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain' }, body: JSON.stringify({ action: 'saveReport', reportData: reportData, username: currentUser.username, ruolo: currentUser.ruolo }) }); const res = await response.json(); if (res.success) { msgEl.style.color = 'green'; msgEl.textContent = res.message; document.getElementById('incidentForm').reset(); document.getElementById('txtScreens').value = ''; document.getElementById('txtCapacity').value = ''; document.getElementById('txtArea').value = ''; } else { msgEl.style.color = 'red'; msgEl.textContent = 'Errore: ' + res.message; } } catch (err) { console.error(err); msgEl.style.color = 'red'; msgEl.textContent = 'Errore di connessione durante il salvataggio.'; } finally { btn.disabled = false; } }
